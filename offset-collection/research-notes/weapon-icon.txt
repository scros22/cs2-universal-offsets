namespace Globals {
    inline bool bESPDroppedWeapons = false;
    inline float espDroppedWeaponsColor[4] = { 1.0f, 1.0f, 1.0f, 1.0f };
    inline void* weaponsFont = nullptr; // ImFont* for weapon icons
}


static const char* GetWeaponIcon(const char* weaponName) {
    // Remove weapon_ prefix for comparison
    const char* cleanName = weaponName;
    if (strncmp(weaponName, "weapon_", 7) == 0) {
        cleanName = weaponName + 7;
    }
    
    // Rifles
    if (strcmp(cleanName, "ak47") == 0) return "W";
    if (strcmp(cleanName, "m4a4") == 0) return "S";
    if (strcmp(cleanName, "m4a1_silencer") == 0) return "T";
    if (strcmp(cleanName, "aug") == 0) return "U";
    if (strcmp(cleanName, "famas") == 0) return "R";
    if (strcmp(cleanName, "galilar") == 0) return "Q";
    if (strcmp(cleanName, "sg556") == 0) return "V";
    
    // Snipers
    if (strcmp(cleanName, "awp") == 0) return "Z";
    if (strcmp(cleanName, "ssg08") == 0) return "a";
    if (strcmp(cleanName, "g3sg1") == 0) return "X";
    if (strcmp(cleanName, "scar20") == 0) return "Y";
    
    // SMGs
    if (strcmp(cleanName, "mac10") == 0) return "K";
    if (strcmp(cleanName, "mp5sd") == 0) return "N";
    if (strcmp(cleanName, "mp7") == 0) return "N";
    if (strcmp(cleanName, "mp9") == 0) return "R";
    if (strcmp(cleanName, "bizon") == 0) return "M";
    if (strcmp(cleanName, "p90") == 0) return "O";
    if (strcmp(cleanName, "ump45") == 0) return "L";
    
    // Shotguns
    if (strcmp(cleanName, "nova") == 0) return "e";
    if (strcmp(cleanName, "sawedoff") == 0) return "c";
    if (strcmp(cleanName, "xm1014") == 0) return "b";
    if (strcmp(cleanName, "mag7") == 0) return "d";
    
    // Heavy
    if (strcmp(cleanName, "m249") == 0) return "g";
    if (strcmp(cleanName, "negev") == 0) return "f";
    
    // Pistols
    if (strcmp(cleanName, "deagle") == 0) return "A";
    if (strcmp(cleanName, "elite") == 0) return "B";
    if (strcmp(cleanName, "fiveseven") == 0) return "C";
    if (strcmp(cleanName, "glock") == 0) return "D";
    if (strcmp(cleanName, "hkp2000") == 0) return "E";
    if (strcmp(cleanName, "usp_silencer") == 0) return "G";
    if (strcmp(cleanName, "p250") == 0) return "F";
    if (strcmp(cleanName, "cz75a") == 0) return "I";
    if (strcmp(cleanName, "tec9") == 0) return "H";
    if (strcmp(cleanName, "revolver") == 0) return "J";
    
    // Utility
    if (strcmp(cleanName, "taser") == 0) return "h";
    if (strcmp(cleanName, "c4") == 0) return "o";
    if (strcmp(cleanName, "knife") == 0 || strcmp(cleanName, "knife_ct") == 0) return "]";
    if (strcmp(cleanName, "knife_t") == 0) return "[";
    
    // Grenades
    if (strcmp(cleanName, "hegrenade") == 0) return "j";
    if (strcmp(cleanName, "flashbang") == 0) return "i";
    if (strcmp(cleanName, "smokegrenade") == 0) return "k";
    if (strcmp(cleanName, "molotov") == 0) return "l";
    if (strcmp(cleanName, "incgrenade") == 0) return "n";
    if (strcmp(cleanName, "decoy") == 0) return "m";
    
    return "?"; // Unknown weapon
}

//
void RenderDroppedWeaponsESP(ImDrawList* draw, int screenWidth, int screenHeight) {
    if (!Globals::bESPDroppedWeapons) return;
    
    // Get local player info
    uintptr_t localPawn = Mem::Read<uintptr_t>(Globals::clientBase + off::dwLocalPlayerPawn);
    if (!localPawn) return;
    
    uintptr_t localSceneNode = Mem::Read<uintptr_t>(localPawn + schema::C_BaseEntity::m_pGameSceneNode);
    Vector3 localOrigin = {};
    if (localSceneNode) {
        localOrigin = Mem::Read<Vector3>(localSceneNode + schema::CGameSceneNode::m_vecAbsOrigin);
    }
    
    uintptr_t entityList = Mem::Read<uintptr_t>(Globals::clientBase + off::dwEntityList);
    if (!entityList) return;
    
    // Loop through entities (starting at 65 to skip players)
    for (int i = 65; i < 8192; i++) {
        uintptr_t listEntry = Mem::Read<uintptr_t>(entityList + 0x10 + 8 * ((i & 0x7FFF) >> 9));
        if (!listEntry) continue;

        uintptr_t entity = Mem::Read<uintptr_t>(listEntry + 0x70 * (i & 0x1FF));
        if (!entity) continue;

        // Get scene node
        uintptr_t sceneNode = Mem::Read<uintptr_t>(entity + schema::C_BaseEntity::m_pGameSceneNode);
        if (!sceneNode) continue;

        // Skip dormant entities
        bool dormant = Mem::Read<bool>(sceneNode + schema::CGameSceneNode::m_bDormant);
        if (dormant) continue;

        // Get entity class name
        uintptr_t entityIdentity = Mem::Read<uintptr_t>(entity + schema::CEntityInstance::m_pEntity);
        if (!entityIdentity) continue;

        uintptr_t designerName = Mem::Read<uintptr_t>(entityIdentity + schema::CEntityIdentity::m_designerName);
        if (!designerName) continue;

        char className[64] = {0};
        for (int c = 0; c < 63; c++) {
            className[c] = Mem::Read<char>(designerName + c);
            if (className[c] == '\0') break;
        }

        // Check if it's a weapon
        bool isWeapon = (strstr(className, "weapon_") != nullptr);
        bool isBomb = (strcmp(className, "weapon_c4") == 0);
        
        if (!isWeapon || isBomb) continue;
        
        // Check if weapon has no owner (dropped)
        // m_hOwnerEntity is a CHandle - 0xFFFFFFFF means invalid/no owner
        uint32_t ownerHandle = Mem::Read<uint32_t>(entity + schema::C_BaseEntity::m_hOwnerEntity);
        
        // Skip if weapon has a valid owner (being held by player)
        if (ownerHandle != 0xFFFFFFFF) continue;

        // Get weapon position
        Vector3 origin = Mem::Read<Vector3>(sceneNode + schema::CGameSceneNode::m_vecAbsOrigin);
        Vector2 screenPos;
        if (!WorldToScreen(origin, screenPos, Globals::viewMatrix, screenWidth, screenHeight)) continue;
        
        // Clean weapon name (remove "weapon_" prefix)
        const char* weaponName = className;
        if (strncmp(className, "weapon_", 7) == 0) {
            weaponName = className + 7;
        }
        
        // Calculate distance
        float distance = (origin - localOrigin).Length() * 0.0254f; // Convert to meters
        
        // Prepare color
        ImU32 weaponCol = IM_COL32(
            (int)(Globals::espDroppedWeaponsColor[0] * 255),
            (int)(Globals::espDroppedWeaponsColor[1] * 255),
            (int)(Globals::espDroppedWeaponsColor[2] * 255),
            (int)(Globals::espDroppedWeaponsColor[3] * 255)
        );
        
        float yOffset = 0.0f;
        
        // Draw weapon icon (if font is loaded)
        const char* weaponIcon = GetWeaponIcon(weaponName);
        if (Globals::weaponsFont) {
            ImGui::PushFont((ImFont*)Globals::weaponsFont);
            
            ImVec2 iconSize = ImGui::CalcTextSize(weaponIcon);
            float iconX = screenPos.x - iconSize.x * 0.5f;
            float iconY = screenPos.y + yOffset;
            
            // Draw icon with black outline for visibility
            draw->AddText(ImVec2(iconX + 1, iconY + 1), IM_COL32(0, 0, 0, 255), weaponIcon);
            draw->AddText(ImVec2(iconX, iconY), weaponCol, weaponIcon);
            
            ImGui::PopFont();
            yOffset += iconSize.y + 2.0f;
        } else {
            // Fallback to text if no weapon font loaded
            ImVec2 textSize = ImGui::CalcTextSize(weaponName);
            float textX = screenPos.x - textSize.x * 0.5f;
            float textY = screenPos.y + yOffset;
            
            draw->AddText(ImVec2(textX + 1, textY + 1), IM_COL32(0, 0, 0, 255), weaponName);
            draw->AddText(ImVec2(textX, textY), weaponCol, weaponName);
            
            yOffset += textSize.y + 2.0f;
        }
        
        // Draw distance below icon
        char distText[32];
        sprintf_s(distText, "%.1fm", distance);
        
        ImVec2 textSize = ImGui::CalcTextSize(distText);
        float textX = screenPos.x - textSize.x * 0.5f;
        float textY = screenPos.y + yOffset;
        
        draw->AddText(ImVec2(textX + 1, textY + 1), IM_COL32(0, 0, 0, 255), distText);
        draw->AddText(ImVec2(textX, textY), weaponCol, distText);
    }
}
