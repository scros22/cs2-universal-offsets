Code:
auto CSkinChanger::ApplyGloves( C_CSPlayerPawn* pLocalPawn , CCSPlayerInventory* pInventory ) -> bool
{
	if ( !pLocalPawn || !pInventory )
		return false;
 
	C_EconItemView* pCurrentGloves = &pLocalPawn->m_EconGloves();
	C_EconItemView* pLoadoutGloves = pInventory->GetItemInLoadout( NormalizeTeam( static_cast<int>( pLocalPawn->m_iTeamNum() ) ) , 41 );
	if ( !pCurrentGloves || !pLoadoutGloves )
		return false;
 
	const bool bNeedsResync =
		( pCurrentGloves->m_iItemID() != pLoadoutGloves->m_iItemID() ) ||
		( pLocalPawn->m_flLastSpawnTimeIndex().m_Value != m_flLastGlovesSpawnTime );
 
	if ( bNeedsResync )
	{
		m_nPendingGloveRefreshFrames = 3;
		pCurrentGloves->m_iItemDefinitionIndex() = pLoadoutGloves->m_iItemDefinitionIndex();
		pCurrentGloves->m_iItemID() = pLoadoutGloves->m_iItemID();
		pCurrentGloves->m_iItemIDHigh() = pLoadoutGloves->m_iItemIDHigh();
		pCurrentGloves->m_iItemIDLow() = pLoadoutGloves->m_iItemIDLow();
		pCurrentGloves->m_iAccountID() = pLoadoutGloves->m_iAccountID();
		pCurrentGloves->pCEconItemDescription() = pLoadoutGloves->pCEconItemDescription();
		m_flLastGlovesSpawnTime = pLocalPawn->m_flLastSpawnTimeIndex().m_Value;
	}
 
	if ( m_nPendingGloveRefreshFrames == 0 )
		return false;
 
	pCurrentGloves->m_bInitialized() = true;
	pLocalPawn->SetBodyGroup();
	pLocalPawn->m_bNeedToReApplyGloves() = true;
 
	--m_nPendingGloveRefreshFrames;
	return true;
}
